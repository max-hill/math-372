\begin{problem}[Exercise 5.1.1]
  A service station has both self-service and full-service
  islands. On each island, there is a single regular unleaded
  pump with two hoses. Let $X$ denote the number of hoses
  being used on the self-service island at a particular time,
  and let $Y$ denote the number of hoses on the full-service
  island in use at that time. The joint pmf of $X$ and $Y$ is described by the
  following table:
  \begin{equation*}
    \begin{array}{c|ccc}
      p(x,y) & y=0 & y=1 & y=2 \\ \hline
      x=0 & 0.10 & 0.08 & 0.06 \\
      x=1 & 0.04 & 0.20 & 0.14 \\
      x=2 & 0.02 & 0.06 & 0.30 \\
    \end{array}
  \end{equation*}
  \begin{enumerate}[(a)]
    \item What is $\P\left[X = 1\text{ and }Y = 1\right]$? 
    \item Compute $\P\left[ X \leq 1 \text{ and } Y \leq 1\right]$?
    \item Using words, describe the event $\left[ X\neq 0 \text{ and } Y\neq 0
    \right]$, and then compute its probability.
    \item Compute the marginal pmf of $X$ and $Y$. Using $p_{X}(x)$, what is
    $\P\left[X \leq 1\right]$?
    \item Are $X$ and $Y$ independent random variables?
  \end{enumerate}
\end{problem}

\begin{problem}[Exercise 5.1.3]
  \label{problem:exercise-5.1.3}
  A grocery store has both an express checkout lane and a superexpress
  checkout lane. Let $X_{1}$ denote the number of customers in line at the
  express checkout at a particular time of day, and let $X_{2}$ denote the
  number of customers in line at the superexpress checkout at the same time.
  Suppose the joint pmf of $X_{1}$ and $X_{2}$ is given by the following
  table:

  \begin{equation*}
    \begin{array}{c|ccccc}
      & x_2=0 & x_2=1 & x_2=2 & x_2=3 & x_2=4 \\ \hline
      x_1=0 & 0.08 & 0.06 & 0.05 & 0.00 & 0.00 \\
      x_1=1 & 0.07 & 0.15 & 0.04 & 0.03 & 0.01 \\
      x_1=2 & 0.04 & 0.05 & 0.10 & 0.04 & 0.05 \\
      x_1=3 & 0.00 & 0.04 & 0.06 & 0.07 & 0.06 \\
    \end{array}
  \end{equation*}
  \begin{enumerate}[(a)]
    \item What $\P\left[X_{1}=1,X_{2}=1 \right]$, that is, the probability
    that there is exactly one customer in each line?
    \item What is $\P\left[X_{1}=X_{2}\right]$, that is, the probability that
    the number of customers in the two lines is identical?
    \item Let $A$ denote the event that there are at least two more customers
    in one line than in the other? Express $A$ in terms of $X_{1}$ and $X_{2}$
    and calculate the probability of this event.
    \item What is the probability that the total number of customers in the
    two lines is exactly four? At least four?
  \end{enumerate}
\end{problem}

\begin{problem}[Exercise 5.1.3]
  Return to the situation described in \cref{problem:exercise-5.1.3}.
  \begin{enumerate}[(a)]
    \item Determine the marginal pmf of $X_{1},$ and then calculate the

    \item Determine the marginal pmf of $X_{2}$.

    $X_{1}$ and $X_{2}$ independent random variables? Explain.
  \end{enumerate}
\end{problem}

\begin{problem}[Exercise 4.3.46]
  The Rockwell hardness of a metal is determined by impressing a hardened
  point into the surface of the metal and then measuring the depth of
  penetration of the point. Suppose the Rockwell hardness of a particular
  alloy is normally distributed with mean 70 and standard deviation 3.

  \begin{enumerate}[(a)]
    \item If a specimen is acceptable only if its hardness is between 67 and
    75, what is the probability that a randomly chosen specimen has an
    acceptable hardness?
    \item If the acceptable range of hardness is $(70 - c, 70 + c)$, for what
    value of $c$ would 95\% of all specimens have acceptable hardness?
    \item  If the acceptable range is as in part (a) and the hardness of each of
    ten randomly selected specimens is independently determined, what is the
    expected number of acceptable specimens among the ten?
    \item  What is the probability that at most eight of ten independently selected
    specimens have a hardness of less than 73.84? [Hint: If $Y$ is the number
    among the ten specimens with hardness less than 73.84, then $Y$ is a
    binomial variable; what is $p$?]
  \end{enumerate}
\end{problem}
