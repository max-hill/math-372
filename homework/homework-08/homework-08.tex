\begin{problem}
  The density function for the standard normal distribution is the function
  \begin{equation*}
    \phi(z)=\frac{1}{\sqrt{2\pi}}e^{-z^{2}/2}, \quad -\infty<z<\infty.
  \end{equation*}
  By computing first and second derivatives, show that $\phi$ is concave down
  for $|z|<1$, concave up for $|z|>1$, with inflection points at $z=\pm 1$.
  Carefully sketch the density curve.
\end{problem}

\begin{problem}[continuous r.v.]
  Let $T$ be the waiting time, in hours, between successive purchases at a
  furniture store. Suppose that $T$ has pdf
  \begin{equation*}
    f_{T}(t) = C e^{-\frac{t}{4}}\mathbf{1}_{\left[ t \geq 0\right] } = \left\{ \begin{array}{l@{\quad:\quad}l} Ce^{-\frac{t}{4}} & t \geq 0\\ 0& t<0\end{array}\right.
  \end{equation*}
  where $C>0$ is some number that you'll have to determine.
  \begin{enumerate}[(a)]
    \item Find a value of $C$ so that $f_{T}(t)$ is a valid pdf.
    \item What is the probability that the time between purchases is greater
    than $6$ minutes? [Note: $6\text{ minutes }= 1/10\text{ of an hour}$].
    \item Compute $\E\left[T \right]$.
  \end{enumerate}
\end{problem}

\begin{problem}
  The Pareto distribution is ``power-law'' distribution used for modeling
  things like income distribution and earthquake magnitudes. Specifically,
  given a random variable $X$, we say that the distribution of $X$ is a member
  of the \demph{Pareto family of distributions} if its cdt is
  \begin{equation*}
    F(x) = \P\left[X \leq x\right] = \left\{ \begin{array}{l@{\quad:\quad}l} 1-\left( \frac{b}{x} \right)^{a}  & x \geq b\\ 0& x<b\end{array}\right.
  \end{equation*}
  for some fixed positive parameters $a$ and $b$.
  \begin{enumerate}[(a)]
    \item Derive the pdf of $X$. Make sure to specify the function on all
    parts of its domain: both when $x \geq b$ and when $x<b$. 
    \item Show that the improper integral
    $\int_{1}^{\infty}\frac{1}{x^{c}}dx$ diverges to $+\infty$ when $c \leq
    1$, but converges to $\frac{1}{c-1}$ when $c>1$. \textit{Hint:} recall that
    \begin{equation*}
      \int_{1}^{\infty}\frac{1}{x^{c}}dx = \lim_{y \to \infty} \int_{1}^{y}  \frac{1}{x^{c}}dx.
    \end{equation*}
    \item What is $\E\left[X \right]$? Consider the cases with $a \leq 1$ and
    $a>0$ separately.
  \end{enumerate}
\end{problem}

\begin{problem}[Exercise 4.1.4]
  Let $X$ denote the vibratory stress (psi) on a wind turbine blade at a
  particular wind speed in a wind tunnel. The article ``Blade Fatigue Life Assessment with
  Application to VAWTS'' \textit{(J. of Solar Energy Engr., 1982)} propose the Rayleigh
  distribution, with pdf
  \begin{equation*}
    f(x) = \left\{ \begin{array}{l@{\quad:\quad}l} \frac{x}{\theta^{2}}e^{-\frac{x^{2}}{2\theta^{2}}} & x>0\\ 0& \text{otherwise}\end{array}\right.
  \end{equation*}
  as a model for the $X$ distribution. (Here, $\theta>0$ is a parameter.)
  \begin{enumerate}[(a)]
    \item Verify that $f(x)$ is a legitimate parameter for any $\theta>0$.
    \item Suppose $\theta=100$ (a value suggested in the article). What is the
    probability that $X$ is at most $200$? Less than 200? At least 200?
    \item What is the probability that $X$ is between 100 and 200 (again
    assuming $\theta=100$)?
    \item Give an expression for $\P\left[X \leq x\right]$.
  \end{enumerate}
\end{problem}
