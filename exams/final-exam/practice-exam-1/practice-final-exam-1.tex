\begin{problem}%[conditional probability]
  An email provider observes that $80\%$ of emails are spam. It intends to filter spam emails before they
  reach the inbox. The filter's accuracy for detecting spam email is $98\%$ and chances of tagging a
  non-spam email as spam is $4\%$.
  \begin{enumerate}[(a)]
    \item Draw a tree diagram to illustrate this situation
    \item If a certain email is tagged as spam, find the probability that it is not a spam email. That
    is, find
    \begin{equation*}
      \P\left[\text{email is not spam}\mid \text{email is tagged as spam} \right]
    \end{equation*}
  \end{enumerate}
\end{problem}



\begin{problem}%[set theory]
  Let $A$ and $B$ be events for which you know $\P\left[A\cup B \right]=0.57$, $\P\left[A \right]=0.22$,
  and $\P\left[A\cap B \right] = 0.16$.
  \begin{enumerate}[(a)]
    \item Find $\P\left[B \right]$ 
    \item Are $A$ and $B$ independent? Justify your answer.
    \item Are $A$ and $B$ disjoint? Justify your answer.
    \item Find $\P\left[A\cap B^{c} \right]$
    \item Find $\P\left[A\mid A\cup B \right]$
    \item Find the probability that exactly one of the two events $A,B$ occurs.
  \end{enumerate}
\end{problem}

\begin{problem}%[conditional probability]
  Two cards are randomly drawn from a deck of cards. Let $A$ be the event that
  at least one ace is drawn. Let $A_{\rm s}$ be the event that the ace of
  spades is chosen. And let $B$ be the event that both cards are aces. Compute
  the following conditional probabilities. You do not need to simplify your
  answers.
  \begin{enumerate}[(a)]
    \item $\P\left[B\mid A_{\rm s} \right]$ 
    \item $\P\left[B\mid A \right]$ 
    \item $\P\left[A_{\rm s} \mid B \right]$ 
  \end{enumerate}
\end{problem}

\begin{problem}[continuous rv]
  A continuous random variable $X$ has the following pdf:
  \begin{equation*}
    f(x) = \left\{ \begin{array}{l@{\quad:\quad}l} 4x^{-5} & x \geq 1\\ 0& x<1\end{array}\right.
  \end{equation*}
  \begin{enumerate}[(a)]
    \item Verify that $f$ is a valid pdf
    \item Find the cumulative density function $F(x)$
    \item Consider a game with a $\$2$ entry fee. The game pays out $X$
    dollars. Find the mean and variance of your net winnings.
  \end{enumerate}
  Now consider playing the game repeatedly, using independent
  observations of $X$. Find the following
  \begin{enumerate}
    \item[(d)] Find the probability that the payout $X$ is greater than $\$10$.
    \item[(e)] What is the probability that the largest payoff in the first six
    plays does not exceed \$10?
    \item[(f)] What is the probability that exactly three of the first six payoffs exceed \$10?
  \end{enumerate}
\end{problem}
