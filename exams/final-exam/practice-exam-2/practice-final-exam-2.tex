\begin{problem}%[combinatorics]
  A dessert company has 32 distinct desserts on its menu: 6 types of pie, 9 types of cake, and 17 flavors of ice
  cream. \textit{(You don't need to simplify your answers for this problem)}
  \begin{enumerate}[(a)]
    \item After purchasing 4 different desserts, a customer decides he will eat them all in a row, and he
    cares about the order in which he eats them. How many different ways can he eat his 4 desserts?
    \item Suppose 7 different desserts are randomly selected from the menu. How many ways are there to do
    this if (i) we care about the order in which the 7 deserts are selected, and (ii) if we don't care
    about the order?
    \item If 9 desserts are randomly selected, how many ways are there to obtain 3 deserts of each type
    (pie, cake, and ice cream)?
    \item If 5 desserts are randomly selected, what's the probability that they are all of the same
    type  (i.e., all pie, all cake, or all ice cream)?
  \end{enumerate}
\end{problem}

\begin{problem}%[chi-squared]
  \phantom{a}
  \vspace{-1em}
  
  \begin{minipage}[b]{0.7\textwidth}
    Morkbork, an alien from the star system Dihedral-5, is testing out some
    cool new \textit{quantum coins} that she recently purchased at the
    interstellar gas station. The packaging advertised that the atoms in the
    coins are ``entangled,'' so that if you flip both coins at the same them,
    they are more likely to land on the same side. Let $H$ and $T$ denote the
    events that the first coin lands on ``heads'' and ``tail''. Let
    $\widetilde{H}$ and $\widetilde{T}$ be the events that the second coin
    lands ``heads'' and ``tails''. Morkbork is skeptical, but she decides to
    test the coins by flipping them together 100 times. She records her
    results in the table:
    \begin{center}
      \begin{tabular}{l|c|c|}
        & $\widetilde{H} $ & $ \widetilde{T} $\\
        \hline
        H& 30& 10\\
        \hline
        T& 20& 40\\
        \hline
      \end{tabular}
    \end{center}
    Morkbork doesn't know if the coins are ``fair'' or not, and she doesn't
    care. She only wants to know if they are ``entangled'' or ``independent''.
    So naturally she decides to do a chi-squared test of independence.
  \end{minipage}%
  \hfill 
  \begin{minipage}[b]{0.34\textwidth}
    \begin{center}
      \vspace{2em}
      \includegraphics[scale=0.10]{../images/morkbork-2}
      \vspace{2em}
    \end{center}
  \end{minipage}

  \begin{enumerate}[(a)]
    \item What is the null hypothesis?
    \item From the table, estimate the probabilities $\P[H ],
    \P[\widetilde{H} ], \P[T ],$ and
    $\P[\widetilde{T} ]$.
    \item Assuming the null hypothesis is true, estimate the probabilities
    \begin{equation*}
      \P[H \cap\widetilde{H} ],\quad \P[H\cap \widetilde{T} ],\quad
      \P[T\cap \widetilde{H} ],\quad\text{ and }\quad\P[T \cap\widetilde{T}].
    \end{equation*}
    \item Based on your answer ot the previous part, make a table of estimated
    expected counts, assuming $H_{0}$.
    \item Compute the $\chi^{2}$ statistic.
    \item Circle the correct phrase to complete the sentence: \textit{A larger
      chi-squared value means the observed data is (very different from / very
      similar to) the expected data.}
    \item Fill in the blank: \textit{In this problem, the chi-squared
      statistic $\chi^{2}$ has a chi-squared distribution with
      \underline{\phantom \quad \quad \quad} degrees of freedom.}
    \item Morkbork's five-year old brother, Glorbmorlk, asks her what
    ``p-value'' means. Help Morkbork give a simple explanation to Glorbmorlk
    by describing in several compete sentences what $p$-value means.
    
    \item Set up, but do not evaluate, an integral to compute a $p$-value
    based on your $\chi^{2}$ statistic. Hint: the density to the chi-squared
    statistic is
    \begin{equation*}
      f(x) = \left\{ \begin{array}{l@{\quad:\quad}l} \frac{1}{\sqrt{2\pi}} x^{-1/2}e^{-x/2} & x \geq 0\\ 0& x<0
        \end{array}\right.
    \end{equation*}
    \item The $p$-value turns out to be a very small number, about
    $3\times 10^{-63}$. How should Morkbork interpret this result?
  \end{enumerate}
\end{problem}

\begin{problem}%[continuous r.v.]
  Let $T$ be the waiting time, in hours, between successive purchases at a
  furniture consignment store. Suppose that $T$ has pdf
  \begin{equation*}
    f_{T}(t) 
    = C e^{-\frac{t}{3}}\mathbf{1}_{\left[ t \geq 0\right] } 
    = \left\{ 
      \begin{array}{l@{\quad:\quad}l} Ce^{-\frac{t}{3}} & t \geq 0\\ 0& t<0\end{array}\right.
  \end{equation*}
  where $C>0$ is some number that you'll have to determine.
  \begin{enumerate}[(a)]
    \item Find a value of $C$ so that $f_{T}(t)$ is a valid pdf.
    \item What is the probability that the time between purchases is greater than $6$ minutes?
    \item Compute $\E\left[T \right]$.
    \item Compute $\Var(T)$. \textit{Hint: integraton by parts}
  \end{enumerate}
\end{problem}




\begin{problem}%[best-fit line]
  Consider the following three data points, of the form $(x,y)$:
  \begin{equation*}
    (-1,4), \quad (0,5), \quad \text{and} \quad (1,9)
  \end{equation*}
  \begin{enumerate}[(a)]
    \item Plot the data points on the $xy$-plane
    \item Suppose we have a line $y=C+Dx$. Let
    $\vec{e}= \begin{bmatrix} e_{1}\\e_{2}\\e_{3} \end{bmatrix}$ be the vector
    whose entries are the vertical distances from the data points to the line.
    Write out $\vec{e}$.
    \item Find the values of $C$ and $D$ which minimize the length of
    $\norm{\vec{e}}^{2}=e_{1}^{2}+e_{2}^{2}+e_{3}^{2}$.
    \item Sketch the line $y=C+Dx$ using your values of $C$ and $D$.
  \end{enumerate}
\end{problem}


\begin{problem}%[combinatorics]
  You are dealt a 5-card poker hand.
  \begin{enumerate}[(a)]
    \item  What is the probability that the fifth
    card you recieve is a spade?
    \item What is the probability that your hand contains at least one spade?
    \item What is the conditional probability that the fifth card you recieve
    is a spade, given that your hand contains at least one spade?
    \item What is the probability that you receive your first spade on the
    third card dealt?
    \item Find the expected number of suits appearing in your hand. (Hint: let
    $S$ be the event that you have at least one spade, and define $D,C,H$
    similarly for diamonds, clubs, and hearts. The number of suits in your
    hand is $N=\mathbf{1}_{S}+\mathbf{1}_{D}+\mathbf{1}_{C}+\mathbf{1}_{H}$.)
  \end{enumerate}
\end{problem}
